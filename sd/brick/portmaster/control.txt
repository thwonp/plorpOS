SDLDBFILE="$controlfolder/gamecontrollerdb.txt"
SDLDBUSERFILE="${HOME}/.config/SDL-GameControllerDB/gamecontrollerdb.txt"
get_controls() {
    DEVICE="0300a3845e0400008e02000014010000"
    param_device="trimuibrick"
    LOWRES="N"
    ANALOGSTICKS="0"
    export SDL_GAMECONTROLLERCONFIG_FILE=/tmp/gamecontrollerdb.txt
    echo "$DEVICE,TRIMUI Player1,a:b1,b:b0,x:b3,y:b2,back:b6,start:b7,guide:b8,leftshoulder:b4,rightshoulder:b5,lefttrigger:a2,righttrigger:a5,dpup:h0.1,dpright:h0.2,dpdown:h0.4,dpleft:h0.8,platform:Linux," > "$SDL_GAMECONTROLLERCONFIG_FILE"
    sdl_controllerconfig="$(< "$SDL_GAMECONTROLLERCONFIG_FILE")"
}
