#!/bin/bash
# plorpOS's PortMaster control.txt on the MagicX boards (Zero 40 plorpos-auv.7,
# Zero 28 plorpos-08c.10), the Brick's (sd/brick/portmaster/control.txt) with
# their pads.
# Unofficial: pm-run.sh copies this over the stock one before every run.
# SPDX-License-Identifier: MIT
CUR_TTY=/dev/null
export controlfolder="/mnt/SDCARD/.userdata/shared/portmaster/PortMaster"
# Ports keep their files in "/$directory/ports/<name>": pm-run.sh's lowercase
# alias of Roms/Ports, on tmpfs.
export directory="tmp/plorpos-roms"
export ESUDO="" ESUDOKILL="-1" ESUDOKILL2="-1"
SDLDBFILE="$controlfolder/gamecontrollerdb.txt"
SDLDBUSERFILE="${HOME}/.config/SDL-GameControllerDB/gamecontrollerdb.txt"
# The pad by its labels, as the Brick's: A is evdev 305 (SDL's b1) and B 304
# (b0), X 308 (b3) and Y 307 (b2) - not launch.sh's map, which is by position.
# The d-pad and L2/R2 are buttons. The Zero 40: one stick, its axes swapped
# by the portrait mount. The Zero 28: R3 after L3 (every later button one up),
# two plain sticks.
BOARD=$(cat /usr/magicx/device 2> /dev/null)
get_controls() {
    DEVICE="1900000012b400006666000000010000"
    param_device="anbernic"
    LOWRES="N"
    export SDL_GAMECONTROLLERCONFIG_FILE=/tmp/gamecontrollerdb.txt
    local pad="$DEVICE,magicx-input,a:b1,b:b0,x:b3,y:b2,back:b8,start:b9,leftshoulder:b4,rightshoulder:b5,lefttrigger:b6,righttrigger:b7,leftstick:b10,"
    if [ "$BOARD" = zero28 ]; then
        ANALOGSTICKS="2"
        pad="${pad}rightstick:b11,dpup:b12,dpleft:b13,dpright:b14,dpdown:b15,guide:b18,leftx:a0,lefty:a1,rightx:a2,righty:a3,"
    else
        ANALOGSTICKS="1"
        pad="${pad}dpup:b11,dpleft:b12,dpright:b13,dpdown:b14,guide:b17,leftx:a1~,lefty:a0,"
    fi
    # The Zero 40 image's pad (launch.sh, MAGICX_PAD): A 304 (b0), B 305 (b1), X 307 (b2),
    # Y 308 (b3), and a 353 at b11 that puts everything after it one up.
    [ "$MAGICX_PAD" = dedicated ] && pad="$DEVICE,magicx-input,a:b0,b:b1,x:b2,y:b3,back:b8,start:b9,leftshoulder:b4,rightshoulder:b5,lefttrigger:b6,righttrigger:b7,leftstick:b10,dpup:b12,dpleft:b13,dpright:b14,dpdown:b15,guide:b18,leftx:a1~,lefty:a0,"
    echo "${pad}platform:Linux," > "$SDL_GAMECONTROLLERCONFIG_FILE"
    sdl_controllerconfig="$(< "$SDL_GAMECONTROLLERCONFIG_FILE")"
}
source "$controlfolder/device_info.txt"
source "$controlfolder/funcs.txt"
# The panel as it is held. The Zero 40's is portrait, 480x800 at 0 degrees.
# PortMaster takes a tall panel for one turned on its side (800x480 at 270),
# and a port sized to that draws past the panel's right edge: GameMaker's patch
# set an 800x480 window and 6 Feet Under came out cropped. Sized to the truth,
# an engine fits its frame to the width, centred. The Zero 28's is 640x480
# landscape (oakMOSS turns the panel). pm-run.sh says it again after any file
# a port script sources (scripts that probe device_info.txt anew), from
# PLORPOS_PM_DISPLAY, which it sets the same way.
if [ "$BOARD" = zero28 ]; then
    PLORPOS_PM_DISPLAY="DISPLAY_WIDTH=640 DISPLAY_HEIGHT=480 DISPLAY_ORIENTATION=0 ASPECT_X=4 ASPECT_Y=3"
else
    PLORPOS_PM_DISPLAY="DISPLAY_WIDTH=480 DISPLAY_HEIGHT=800 DISPLAY_ORIENTATION=0 ASPECT_X=3 ASPECT_Y=5"
fi
[ -n "${PLORPOS_PM_GUI:-}" ] || export $PLORPOS_PM_DISPLAY
# Ports are told ROCKNIX, as on the Brick - but not Westonpack's wrapper,
# which sources this too: named ROCKNIX it expects ROCKNIX's sway and has its
# Crusty present over Wayland ("wayland not available"); under the probed
# name it brings its own Weston and Crusty draws through SDL (the GKD's way).
# device_info.txt does not probe again under a port's environment, so the
# name oakMOSS probes as is said here.
case "$0" in
*/westonwrap.sh) export CFW_NAME=Unknown ;;
*) [ -n "${PLORPOS_PM_GUI:-}" ] || export CFW_NAME=ROCKNIX ;;
esac
export GPTOKEYB="$controlfolder/gptokeyb $ESUDOKILL"
export GPTOKEYB2="env LD_PRELOAD=$controlfolder/libinterpose.${DEVICE_ARCH}.so $controlfolder/gptokeyb2 $ESUDOKILL2"
