#!/bin/bash
#
# plorpOS's PortMaster control.txt on the GKD 350H Ultra under the BasedOS fork
# (plorpos-gkd.94.15). Unofficial: pm-run.sh copies this over the stock one
# before every run. The H700's (sd/h700/portmaster/control.txt), BasedOS being
# no CFW PortMaster knows either, with the GKD's pad from the ROCKNIX build's
# (sd/gkd/portmaster/control.txt).
#
# SPDX-License-Identifier: MIT

CUR_TTY=/dev/null

export controlfolder="/mnt/SDCARD/.userdata/shared/portmaster/PortMaster"
# Ports keep their files in "/$directory/ports/<name>": Roms/Ports on the card
# (exFAT looks names up case-blind).
export directory="mnt/SDCARD/Roms"

# Root already, and no sudo on BasedOS.
export ESUDO=""
export ESUDOKILL="-1"
export ESUDOKILL2="-1"

SDLDBFILE="$controlfolder/gamecontrollerdb.txt"
SDLDBUSERFILE="${HOME}/.config/SDL-GameControllerDB/gamecontrollerdb.txt"

# The pad by the GUID SDL 2.32 gives it, which carries a CRC of its name, with
# labels as on the case and the launcher (platform_gkd.c): A east, B south,
# L1/R1 BTN_TL/TR, L2/R2 BTN_TL2/TR2. Without a DEVICE the stock get_controls
# hands every port the whole database.
get_controls() {
  DEVICE="1900c3cb010000000321000000010000"
  param_device="gkd"
  LOWRES="N"
  ANALOGSTICKS="1"
  export SDL_GAMECONTROLLERCONFIG_FILE="/tmp/gamecontrollerdb.txt"
  echo "$DEVICE,gkd_atom_joypad,b:b0,a:b1,x:b2,y:b3,leftshoulder:b4,rightshoulder:b5,lefttrigger:b6,righttrigger:b7,back:b8,start:b9,guide:b10,leftstick:b11,dpup:b12,dpdown:b13,dpleft:b14,dpright:b15,leftx:a0,lefty:a1,platform:Linux," \
    > "$SDL_GAMECONTROLLERCONFIG_FILE"
  sdl_controllerconfig="$(< "$SDL_GAMECONTROLLERCONFIG_FILE")"
}

source $controlfolder/device_info.txt
source $controlfolder/funcs.txt

# Ports run bare on KMS's GL, as on the H700, and PortMaster's ROCKNIX branch
# is the one that skips its own compositor. Ports only: PortMaster's own GUI
# stays generic (pm-run.sh sets PLORPOS_PM_GUI). So does Westonpack's wrapper,
# which sources this too: named ROCKNIX it expects ROCKNIX's sway already
# running, while its generic mode brings its own Weston and draws through
# Crusty on KMS, which is what a system with no compositor needs.
case "$0" in
*/westonwrap.sh) ;;
*) [ -n "${PLORPOS_PM_GUI:-}" ] || export CFW_NAME="ROCKNIX" ;;
esac

export GPTOKEYB2="$ESUDO env LD_PRELOAD=$controlfolder/libinterpose.${DEVICE_ARCH}.so $controlfolder/gptokeyb2 $ESUDOKILL2"
export GPTOKEYB="$ESUDO $controlfolder/gptokeyb $ESUDOKILL"
